# Keep the Leonardo env on a known-compatible audio/CUDA stack.
#
# Constraints (lowest common denominator):
#   * pyannote.audio 3.x needs torchaudio.AudioMetaData (present in 2.3-2.8,
#     removed in 2.9).
#   * torbi is a compiled extension. Version 1.4.0 only publishes a Linux wheel
#     for glibc 2.34+, which is too new for Leonardo, and older source releases
#     must be built against the same torch that imports them.
#   * transformers 5.x needs torch >= 2.4 too but is brand new; pin to v4.x
#     since we only use standard AutoModel/AutoTokenizer APIs.
# => torch 2.5.1 is the newest cu121 wheel and satisfies all three.
torch==2.5.1
torchaudio==2.5.1
pyannote.audio==3.3.0
transformers>=4.45,<5
penn==1.0.0
torbi==1.3.3

# Current bitsandbytes releases require newer torch versions. 0.43.1 is the
# version this setup originally targeted for 4-bit loading on A100.
bitsandbytes==0.43.1

# Older torch wheels were compiled against NumPy 1.x and can raise
# "module compiled using NumPy 1.x cannot be run in NumPy 2.x" at import time
# if pip pulls in numpy 2.x via any transitive dep (transformers, datasets, etc.).
numpy<2
